function updateLivePrice() { const model = document.getElementById("modelOpt").value; const size = document.getElementById("sizeOpt").value; const care = document.getElementById("careOpt").value; const term = document.getElementById("termOpt").value; const rType = document.getElementById("rentalTypeOpt").value; const data = rawDB[currentTab][model].sizes[size][care][term][rType]; const box = document.getElementById("priceDisplayBox"); // 💡 데이터가 기존 객체 형태이든, 새로 바뀐 순수 원가(숫자) 형태이든 상관없이 기준 금액을 빼옵니다. const basePrice = (typeof data === 'object') ? (data.promo || data.catalog) : data; // 💡 이전의 복잡했던 가짜 할인 렌더링을 없애고, 원가를 직관적으로 보여준 뒤 // 할인 계산은 온전히 장바구니(renderCart) 로직에 맡깁니다. let html = ""; if(currentTab === "bed" || currentTab === "healing") { html = `
기본 렌탈료 (원가)${basePrice.toLocaleString()}원
`; } else { html = `
기본 렌탈료 (원가)${basePrice.toLocaleString()}원
* 장바구니에 담으면 프로모션 할인이 자동 계산됩니다.
`; } box.innerHTML = html; // 장바구니로 넘어갈 기준 금액 세팅 document.getElementById("dispFinal").dataset.promo = basePrice; document.getElementById("dispFinal").dataset.term = term; document.getElementById("dispFinal").dataset.rtype = rType; }
